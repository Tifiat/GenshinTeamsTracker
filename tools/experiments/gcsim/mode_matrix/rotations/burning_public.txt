options swap_delay=12 iteration=1000;
target lvl=100 resist=0.1 radius=2 pos=0,2.5 hp=999999999;
energy every interval=480,720 amount=1;

active kinich;
emilie skill;
for let i=0; i<4; i=i+1 {
  if i > 0 && .dehya.skill.cd {
    bennett skill;
  }
  dehya skill:2;
  bennett skill, burst;
  if i {
    if .emilie.burst.ready {
      emilie burst;
    } else {
      emilie skill, attack;
    }
  }
  kinich skill;
  for let c=0; c<5; c=c+1 {
    kinich attack[direction=1]:2;
    while .kinich.nightsoul.points < 20 && .kinich.nightsoul.state {
      wait(1);
    }
    if .kinich.nightsoul.state {
      kinich skill[hold=1];
    }
  }
  if .kinich.burst.ready {
    kinich burst;
  }
}
