options swap_delay=12 iteration=1000;
target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;

active ineffa;
for let i=0; i<6; i=i+1 {
  ineffa skill;
  if !is_even(i) {
    ineffa burst;
  } else {
    ineffa attack;
  }
  columbina skill;
  if is_even(i) {
    columbina burst;
  } else {
    columbina attack;
  }
  if .sucrose.burst.ready {
    sucrose burst, attack, skill, dash;
  } else {
    sucrose attack, skill, dash;
  }
  flins skill,
        skill, attack, burst,
        attack:3, dash, attack:3, dash, attack:3, dash, attack,
        skill, burst,
        attack:3, dash, attack:3, dash, attack:3, dash, attack;
  if .flins.status.manifest-flame {
    flins skill;
    if .flins.energy >= 30 {
      flins burst;
    }
  }
}
