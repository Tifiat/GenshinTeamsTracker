fn nahida_normal_attack_spam() {
  let NA = 0;
  while !.nahida.burst.ready {
    if NA == 4 {
      nahida dash;
      NA = 0;
    }
    if !.nahida.burst.ready {
      nahida attack;
      NA = NA + 1;
    }
  }
}

active nahida;
for let i=0; i<4; i=i+1 {
  nahida burst, skill;
  fischl attack:1;
  if .fischl.skill.ready {
    fischl skill;
  } else {
    fischl burst;
  }
  kuki skill;
  tighnari skill, aim:3, burst;
  kuki burst;
  nahida skill, attack, charge, dash, attack:2, charge, dash, attack:2, charge, dash;
  nahida_normal_attack_spam();
}
