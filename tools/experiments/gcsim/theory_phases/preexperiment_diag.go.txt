type preparationCoverage struct {Owner int;Kind string;Known,Unknown int}

type preparationInput struct {Seed string;WorldStates int;FirstOrder uint64;WorldPresent,WorldComplete bool;StatesWithFacts,StatesWithCustom int;OrderOneIndex int;OrderOneOrder uint64;OrderOneFacts,OrderOneCustom bool}
type PreparationDiagnostic struct {

func (b *NeutralBinding) preparationReplayObserved(choices []RuleChoice,member neutralBoundMember,symbolic bool,r RuleReplay,seconds float64){
 d:=b.preparationDiagnostic;if d==nil{return};started:=time.Now();defer func(){d.ObserverSeconds+=time.Since(started).Seconds()}()

 }
 for i,value:=range r.Values {
  if i<len(member.ports){q:=member.ports[i].query;key:=fmt.Sprintf("%d|%s",q.Owner,q.Kind);index,ok:=coverage[key];if !ok{index=len(row.Coverage);coverage[key]=index;row.Coverage=append(row.Coverage,preparationCoverage{Owner:q.Owner,Kind:q.Kind})};if value.Known{row.Coverage[index].Known++}else{row.Coverage[index].Unknown++}}
  if value.Known{continue};row.UnknownPorts++

  key:=fmt.Sprintf("%d|%s|%s|%s|%s",q.Owner,q.Kind,q.Stat,q.Mode,value.Boundary)
  if index,ok:=unknown[key];ok{row.Unknown[index].Count++;continue}
  if len(unknown)>=8{row.Overflow++;continue};unknown[key]=len(row.Unknown)
  // Keep query provenance only; large Attack/State/keys payloads stay in tape.
  sample:=RuleQuery{HitID:q.HitID,Order:q.Order,Frame:q.Frame,Active:q.Active,Owner:q.Owner,Target:q.Target,Kind:q.Kind,Mode:q.Mode,Stat:q.Stat}

 }

}

// Exactly two passes of <=16 explicitly selected package cases. No search,
// allocation, ordinary verification, altered frozen policy, or capture here.
func (b *NeutralBinding) DiagnosePreparation(cases []PreparationCase,bind func(NeutralExpressions)PreparationBindingEvidence)(*PreparationDiagnostic,error){
 if b==nil||len(b.members)==0||len(cases)==0||len(cases)>16||b.preparationDiagnostic!=nil{return nil,fmt.Errorf("invalid bounded preparation diagnostic")}
 d:=&PreparationDiagnostic{Schema:"theory_preparation_events_v1"};b.preparationDiagnostic=d;defer func(){b.preparationDiagnostic=nil}()
 for _,member:=range b.members {row:=preparationInput{Seed:member.seed,OrderOneIndex:-1};world:=member.neutral.World;if world!=nil{row.WorldPresent=true;row.WorldStates=len(world.States);row.WorldComplete=world.Complete;for i,state:=range world.States{if i==0{row.FirstOrder=state.Order};if state.Facts!=nil{row.StatesWithFacts++;if _,ok:=state.Facts.Fields["Custom"];ok{row.StatesWithCustom++}};if state.Order<=1{row.OrderOneIndex=i;row.OrderOneOrder=state.Order;row.OrderOneFacts=state.Facts!=nil;row.OrderOneCustom=false;if state.Facts!=nil{_,row.OrderOneCustom=state.Facts.Fields["Custom"]}}}};d.Inputs=append(d.Inputs,row)}
 owners:=map[string]int{};for i,c:=range b.members[0].neutral.Context.Characters {owners[c.Key]=i}
 choices:=make([][]RuleChoice,len(cases));ids:=map[string]bool{}
 for i,test:=range cases {
  if test.ID==""||ids[test.ID]{return nil,fmt.Errorf("diagnostic case ID")};ids[test.ID]=true
  for _,pkg:=range test.Packages {owner,ok:=owners[pkg.Wearer];if !ok{return nil,fmt.Errorf("unknown diagnostic wearer")};for _,set:=range pkg.Sets{choices[i]=append(choices[i],RuleChoice{owner,set.Key,set.Pieces})}}
 }
 for pass:=1;pass<=2;pass++ {for i,test:=range cases {
  d.current,d.pass=test.ID,pass;d.frozenCoverage=map[preparationFrozenKey]preparationFrozenCoverage{}
  started:=time.Now();out:=b.CompilePackagesSparse(choices[i]);elapsed:=time.Since(started).Seconds()
  observe:=time.Now();hash:=sha256.New()
  // This diagnostic output digest is NOT a compiled arithmetic-family identity.
  if err:=json.NewEncoder(hash).Encode(struct{Output NeutralExpressions;Frozen bool}{out,out.r1Frozen});err!=nil{return nil,err}
  _,identityAvailable:=out.PreparedPayloadIdentity();stats:=b.R1FrozenStats();stats.Records=nil;stats.PackageGuides=nil
  binding:=PreparationBindingEvidence{};if bind!=nil{binding=bind(out)}
  frozen:=[]preparationFrozenCoverage{};for _,row:=range d.frozenCoverage{frozen=append(frozen,row)};sort.Slice(frozen,func(i,j int)bool{a,z:=frozen[i],frozen[j];if a.Seed!=z.Seed{return a.Seed<z.Seed};if a.Owner!=z.Owner{return a.Owner<z.Owner};return a.Kind<z.Kind})
  d.Cases=append(d.Cases,preparationCaseResult{test.ID,pass,ruleChoicesKey(choices[i]),elapsed,out.Complete(),out.r1Frozen,out.footprint.trackedStats(),identityAvailable,hex.EncodeToString(hash.Sum(nil)),len(out.Constants),len(out.SparseEffects),preparationStrings(out.Boundaries),preparationStrings(out.Uncertainty),b.PreparationStats(),stats,binding,frozen})
  d.ObserverSeconds+=time.Since(observe).Seconds()-binding.Seconds
 }}
 return d,nil
}
