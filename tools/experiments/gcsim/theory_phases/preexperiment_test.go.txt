package seteffects

import("testing";"reflect";"encoding/json";"os";"genshinteamstracker/native/gcsim_optimizer/internal/contracts")

func TestPreparationDiagnosticObserverParity(t *testing.T){
 var bundle SourceBundle;var request struct{Engine contracts.EngineBinding}
 read:=func(path string,value any){raw,err:=os.ReadFile(path);if err!=nil{t.Fatal(err)};if err=json.Unmarshal(raw,value);err!=nil{t.Fatal(err)}}
 read(os.Getenv("GTT_RULE_SOURCE_BUNDLE"),&bundle);read(os.Getenv("GTT_RULE_REQUEST"),&request)
 c,err:=CompileNeutralTheorySources(bundle,request.Engine,NewRuleParseCache("",""));if err!=nil{t.Fatal(err)}
 n:=ruleTestNeutral()
 // Existing fixture constants, resolved through the current lazy repository
 // API instead of the legacy helper's already-loaded-module assumption.
 constant:=func(module,name string)float64{for _,s:=range c.Sets{if s.rule!=nil{m:=s.rule.module(gcsimImport+module);if m!=nil{v:=newRuleVM(s.rule,nil,1000);out,err:=ruleTry(func()any{return v.lookup(m.envs[m.files[0]],name)});if err!=nil{t.Fatal(err)};return ruleNumber(out)}}};t.Fatal("fixture source constant absent",module,name);return 0}
 enemyDamage:=int(constant("pkg/core/event","OnEnemyDamage"));lunarEvent:=int(constant("pkg/core/event","OnLunarCharged"))
 skill:=constant("pkg/core/attacks","AttackTagElementalArt");hydro:=constant("pkg/core/attributes","Hydro")
 first:=ruleTestEvent(enemyDamage,2,skill,hydro,1);second:=ruleTestEvent(lunarEvent,2,skill,hydro,1);second.Order=12;second.EndOrder=13
 third:=ruleTestEvent(enemyDamage,0,skill,hydro,1);third.Order=14;third.EndOrder=15;n.Ledger.Events=[]contracts.CombatEvent{first,second,third}
 for i,stat:=range []string{"em","cr","atk%"}{n.Stats=append(n.Stats,contracts.NeutralStatInput{NodeID:uint32(i+1),Owner:"beta",Kind:"stat",Stat:stat,Point:contracts.NeutralInputPoint{Mode:"snapshot",Order:20,Frame:11}})}
 makeBinding:=func(observer bool)*NeutralBinding{b,err:=NewNeutralBinding(c,[]contracts.EffectInputsOutput{{SourceManifestBodySHA256:c.SourceManifestSHA256,Neutral:n}});if err!=nil{t.Fatal(err)};b.EnableR1FrozenResiduals();if observer{b.preparationDiagnostic=&PreparationDiagnostic{current:"fixture",pass:1}};return b}
 off,on:=makeBinding(false),makeBinding(true)
 cases:=[][]RuleChoice{{{1,"nightoftheskysunveiling",4},{2,"silkenmoonsserenade",4}},{{0,"silkenmoonsserenade",4},{2,"silkenmoonsserenade",4}},{{0,"tenacityofthemillelith",4},{1,"nightoftheskysunveiling",2},{1,"emblemofseveredfate",2}},{{0,"noblesseoblige",4},{2,"noblesseoblige",4},{1,"nightoftheskysunveiling",4}},{{0,"berserker",4}},{{0,"oceanhuedclam",2},{0,"emblemofseveredfate",2}}}
 for pass:=0;pass<2;pass++{for _,choices:=range cases{
  a,b:=off.CompilePackagesSparse(choices),on.CompilePackagesSparse(choices)
  if !reflect.DeepEqual(a,b){t.Fatal("observer changed arithmetic/output",choices)}
  x,y:=off.PreparationStats(),on.PreparationStats();x.ReplaySeconds,y.ReplaySeconds=0,0
  if !reflect.DeepEqual(x,y)||!reflect.DeepEqual(off.R1FrozenStats(),on.R1FrozenStats()){t.Fatal("observer changed integer counters/frozen records")}
 }}
 if len(on.preparationDiagnostic.Joint)==0||len(on.preparationDiagnostic.FreezeConcrete)==0{t.Fatal("fixture did not exercise diagnostic joint/freeze hooks")}
}

func TestPreparationDiagnosticBoundedObservations(t *testing.T){
 b:=r1TestBinding();d:=&PreparationDiagnostic{current:"fixture",pass:1};b.preparationDiagnostic=d
 r:=RuleReplay{Values:[]RuleValue{{Known:false,Boundary:"unobserved"}}}
 for i:=0;i<20;i++{r.MechanicBoundaries=append(r.MechanicBoundaries,contracts.MechanicBoundary{MechanicUID:"fixture",Reason:"unobserved"})}
 b.preparationReplayObserved(nil,b.members[0],true,r,0.25)
 if len(d.Replays)!=1||d.Replays[0].Causes[0].Count!=20||d.Replays[0].UnknownPorts!=1||d.Replays[0].Seconds!=0.25{t.Fatal("observation grouping")}
 for i:=0;i<520;i++{b.preparationReplayObserved(nil,b.members[0],false,r,0)}
 if len(d.Replays)!=512||d.Dropped!=9{t.Fatal("finite rows",len(d.Replays),d.Dropped)}
 b.preparationDiagnostic=nil;b.preparationReplayObserved(nil,b.members[0],false,r,1)
 if len(d.Replays)!=512{t.Fatal("disabled observer changed records")}
}
